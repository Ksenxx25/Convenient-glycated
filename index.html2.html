#include int main() { int n; float sum = 0.0, value, average; // Запрос количества элементов printf("Введите количество показателей: "); if (scanf("%d", &n) != 1 || n <= 0) { printf("Ошибка: введено некорректное количество.\n"); return 1; } // Ввод самих показателей for (int i = 1; i <= n; ++i) { printf("Введите показатель %d (в %%): ", i); scanf("%f", &value); sum += value; // Суммируем значения } // Расчет среднего значения average = sum / n; // Вывод результата с точностью до двух знаков после запятой printf("\nСредний показатель: %.2f%%\n", average); return 0; }
